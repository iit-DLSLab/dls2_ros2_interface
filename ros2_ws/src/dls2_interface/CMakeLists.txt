# print_hashes: prints the RIHS type hash of every message, one per line as
# "dls2_interface/msg/<Name>: RIHS01_<hex>".
# Usage: ros2 run dls2_interface print_hashes
find_package(rosidl_typesupport_cpp REQUIRED)
find_package(rosidl_runtime_c REQUIRED)

set(PRINT_HASHES_INCLUDES "")
set(PRINT_HASHES_CALLS "")
foreach(msg_file ${msg_files})
  get_filename_component(msg_name ${msg_file} NAME_WE)
  # CamelCase -> snake_case, same rule rosidl uses for the generated header names
  string(REGEX REPLACE "(.)([A-Z][a-z]+)" "\\1_\\2" msg_header "${msg_name}")
  string(REGEX REPLACE "([a-z0-9])([A-Z])" "\\1_\\2" msg_header "${msg_header}")
  string(TOLOWER "${msg_header}" msg_header)
  string(APPEND PRINT_HASHES_INCLUDES "#include \"${PROJECT_NAME}/msg/${msg_header}.hpp\"\n")
  string(APPEND PRINT_HASHES_CALLS
    "  print_hash<${PROJECT_NAME}::msg::${msg_name}>(\"${PROJECT_NAME}/msg/${msg_name}\");\n")
endforeach()

# file(CONFIGURE) rewrites the file only when its content changes
file(CONFIGURE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/print_hashes.cpp" @ONLY CONTENT [=[
// Generated by CMakeLists.txt: do not edit.
#include <iomanip>
#include <iostream>
#include <sstream>
#include <string>

#include <rosidl_runtime_c/type_hash.h>
#include <rosidl_typesupport_cpp/message_type_support.hpp>

@PRINT_HASHES_INCLUDES@
template<class MsgT>
void print_hash(const char * type_name)
{
  const auto * ts = rosidl_typesupport_cpp::get_message_type_support_handle<MsgT>();
  const auto * h = ts->get_type_hash_func(ts);

  std::ostringstream out;
  out << "RIHS" << std::setw(2) << std::setfill('0') << static_cast<int>(h->version) << "_";
  for (const auto b : h->value) {
    out << std::hex << std::setw(2) << std::setfill('0') << static_cast<int>(b);
  }
  std::cout << type_name << ": " << out.str() << std::endl;
}

int main()
{
@PRINT_HASHES_CALLS@
  return 0;
}
]=])

add_executable(print_hashes "${CMAKE_CURRENT_BINARY_DIR}/print_hashes.cpp")
rosidl_get_typesupport_target(cpp_typesupport_target ${PROJECT_NAME} rosidl_typesupport_cpp)
target_link_libraries(print_hashes
  ${cpp_typesupport_target}
  rosidl_typesupport_cpp::rosidl_typesupport_cpp
  rosidl_runtime_c::rosidl_runtime_c
)
install(TARGETS print_hashes
  DESTINATION lib/${PROJECT_NAME}
)
